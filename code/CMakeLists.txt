cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
endif()
project(ra2_native VERSION 0.1.0 LANGUAGES CXX)
include(cmake/CompilerOptions.cmake)
include(cmake/CheckYrppLayout.cmake)
include(cmake/CoreIncludes.cmake)

option(RA2_BUILD_GODOT "Build the optional Godot GDExtension" ON)
option(BUILD_TESTING "Build product tests" OFF)
include(CTest)
include(cmake/Core.cmake)
include(cmake/Render.cmake)
option(RA2_BUILD_MOVIES "Enable the optional FFmpeg decoder in the core" ON)
if(RA2_BUILD_MOVIES)
    include(cmake/Movies.cmake)
else()
    target_sources(ra2_core PRIVATE core/src/movie_activity_disabled.cpp)
endif()
option(RA2_BUILD_SOFTWARE_RENDER "Build the optional YR software rasterizer (not a Godot dependency)" OFF)
if(RA2_BUILD_SOFTWARE_RENDER)
    include(cmake/SoftwareRender.cmake)
endif()
if(RA2_BUILD_GODOT)
    include(cmake/Godot.cmake)
endif()
if(BUILD_TESTING)
    include(cmake/Tests.cmake)
endif()
